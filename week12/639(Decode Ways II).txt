class Solution:
    def numDecodings(self, s: str) -> int:
        MOD= 10**9+7
        #只看一個數字
        def decode_one(c:str) -> int:
            if c=='*':return 9
            if c=='0' :return 0
            return 1
        #看兩個數字
        def decode_two(c:str , d:str) -> int:
            if c=='*':
                if d=='*':return 15
                elif d<='6' :return 2
                else: return 1
            elif c=='1':
                if d=='*':return 9
                return 1
            elif c=='2':
                if d=='*': return 6
                elif d<='6': return 1
                else:return 0
            return 0
        
        
        pre2=1 #紀錄上上一格，初始值是1
        pre1=decode_one(s[0])#紀錄上一格方法總數，初始值是s[0]

        for i in range(1,len(s)):
            #看一個數
            way1=pre1*decode_one(s[i])
            #看兩個數
            way2=pre2*decode_two(s[i-1],s[i])

            ways=(way1+way2)%MOD

            #更新狀態
            pre2=pre1
            pre1=ways

        return pre1
        '''
        時間複雜度：O(N)
        空間複雜度：O(1)
        '''
