class Solution:
    def combinationSum4(self, nums: List[int], target: int) -> int:
        #直接判斷沒辦法達成的
        if min(nums)> target:
            return 0
        # 建立一個大小是 target + 1 的 dp 陣列，先填 0
        dp = [0] * (target + 1)
        
        # Base case
        dp[0] = 1
        
        #從 1 算到 target
        for i in range(1, target + 1):
            
            # 內層迴圈：嘗試手邊的每一個選項 (步伐)
            for num in nums:
                
                # 選項的數字不能大於目前的目標
                if i >= num:
                    
                    # 把過往的總和加總起來
                    dp[i] += dp[i - num]
                    
        return dp[target]
        '''
        時間複雜度:O(target*N)=O(N)
        空間複雜度:O(target)
        '''
