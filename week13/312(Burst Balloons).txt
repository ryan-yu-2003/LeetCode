class Solution:
    def maxCoins(self, nums: List[int]) -> int:
        #反向思考去看誰當最後一個被打破的氣球分數會最高

        val=[1]+nums+[1]#依照題目把其中沒邊界的地方邊界放1
        @cache
        def dsp(i:int,j:int) ->int :
            #窮舉誰當最後一個被射破的氣球
            maxscore=0
            if i>= j-1:
                return 0
            for k in range(i+1,j):
                score=dsp(i,k)+dsp(k,j)+val[k]*val[i]*val[j]
                maxscore=max(maxscore,score)

            return maxscore
        return dsp(0,len(val)-1)   
        '''
        時間複雜度：O(N^3)
        空間複雜度：O(N^2)
        '''
